﻿import React, { useState } from 'react';
import { 
  Droplets, Save, Calendar, Search, 
  ChevronRight, TrendingUp, Filter, Plus,
  Award, Zap
} from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';
 
const MilkControl: React.FC = () => {
  const [liters, setLiters] = useState<string>('');
  const [selectedAnimal, setSelectedAnimal] = useState<string | null>(null);
 
  // Mock data for the view
  const cows = [
    { id: 'HEM-1024', name: 'Mariposa', last: '12.5L', status: 'Lactante' },
    { id: 'HEM-1056', name: 'Estrella', last: '14.0L', status: 'Lactante' },
    { id: 'HEM-982', name: 'Doris', last: '11.2L', status: 'Lactante' },
  ];
 
  return (
    <div className="p-4 md:p-8 bg-slate-50 min-h-screen font-sans">
      <header className="mb-8">
        <h1 className="text-3xl font-black text-slate-900 tracking-tight">Control de Pesaje</h1>
        <p className="text-slate-500 font-medium flex items-center gap-2 mt-1">
          <Droplets className="w-5 h-5 text-blue-500" /> Registro diario de producción láctea
        </p>
      </header>
 
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
        {/* Cow Selector (Quick List) */}
        <div className="lg:col-span-1 space-y-4">
          <div className="relative mb-6">
            <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-slate-400" />
            <input 
              type="text" 
              placeholder="Buscar vaca..."
              className="w-full bg-white border-2 border-slate-100 rounded-2xl py-4 pl-12 pr-4 font-bold text-slate-700 outline-none focus:border-blue-500 transition-all shadow-sm"
            />
          </div>
          
          <div className="space-y-3">
            {cows.map((cow) => (
              <motion.div 
                whileTap={{ scale: 0.98 }}
                onClick={() => setSelectedAnimal(cow.id)}
                key={cow.id}
                className={`p-5 rounded-3xl cursor-pointer border-2 transition-all flex justify-between items-center ${
                  selectedAnimal === cow.id 
                    ? 'bg-blue-600 border-blue-600 text-white shadow-xl shadow-blue-100' 
                    : 'bg-white border-white hover:border-blue-200 text-slate-800 shadow-sm'
                }`}
              >
                <div>
                  <h4 className="font-black">{cow.id}</h4>
                  <p className={`text-xs font-bold ${selectedAnimal === cow.id ? 'text-blue-100' : 'text-slate-400'}`}>
                    {cow.name} • Último: {cow.last}
                  </p>
                </div>
                <ChevronRight className={`w-5 h-5 ${selectedAnimal === cow.id ? 'text-white' : 'text-slate-300'}`} />
              </motion.div>
            ))}
          </div>
        </div>
 
        {/* Input Panel (The "Keypad" style) */}
        <div className="lg:col-span-2">
          <AnimatePresence mode="wait">
            {selectedAnimal ? (
              <motion.div 
                initial={{ opacity: 0, y: 20 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -20 }}
                className="bg-white p-8 md:p-12 rounded-[3.5rem] shadow-xl border border-slate-100 h-full flex flex-col"
              >
                <div className="flex justify-between items-start mb-8">
                  <div>
                    <span className="bg-blue-50 text-blue-600 px-3 py-1 rounded-full text-[10px] font-black uppercase tracking-widest">Registrando</span>
                    <h2 className="text-4xl font-black text-slate-900 mt-2">{selectedAnimal}</h2>
                  </div>
                  <div className="text-right">
                    <p className="text-xs font-black text-slate-400 uppercase">Promedio Mes</p>
                    <p className="text-2xl font-black text-emerald-500">12.8 L</p>
                  </div>
                </div>
 
                <div className="flex-1 flex flex-col items-center justify-center py-8">
                  <div className="relative mb-12">
                     <input 
                       type="number"
                       value={liters}
                       onChange={(e) => setLiters(e.target.value)}
                       placeholder="00.0"
                       className="text-[8rem] font-black text-slate-900 bg-transparent border-none outline-none text-center w-full focus:ring-0 placeholder:text-slate-100"
                     />
                     <span className="absolute -right-12 bottom-8 text-4xl font-black text-slate-300">Litros</span>
                  </div>
 
                  <div className="grid grid-cols-3 gap-4 w-full max-w-sm">
                    {[1, 2, 3, 4, 5, 6, 7, 8, 9, '.', 0].map(n => (
                      <button 
                        key={n}
                        onClick={() => setLiters(prev => prev + n.toString())}
                        className="h-16 bg-slate-50 rounded-2xl font-black text-2xl text-slate-700 hover:bg-slate-100 active:scale-95 transition-all"
                      >
                        {n}
                      </button>
                    ))}
                    <button 
                      onClick={() => setLiters('')}
                      className="h-16 bg-red-50 rounded-2xl font-black text-lg text-red-600 hover:bg-red-100"
                    >
                      Borrar
                    </button>
                  </div>
                </div>
 
                <button className="w-full bg-blue-600 text-white p-6 rounded-[2rem] font-black text-xl flex items-center justify-center gap-4 shadow-2xl shadow-blue-100 hover:bg-blue-700 transition-all active:scale-95 mt-8">
                  <Save className="w-8 h-8" /> Guardar Pesaje
                </button>
              </motion.div>
            ) : (
              <div className="bg-slate-100/50 border-4 border-dashed border-slate-200 rounded-[3.5rem] h-full flex flex-col items-center justify-center text-slate-400 p-12 text-center">
                <Droplets className="w-24 h-24 mb-6 opacity-10" />
                <h3 className="text-2xl font-black text-slate-400">Seleccione una vaca</h3>
                <p className="font-bold max-w-xs mt-2 opacity-60">Toque una vaca de la lista de la izquierda para ingresar su pesaje de hoy.</p>
              </div>
            )}
          </AnimatePresence>
        </div>
      </div>
    </div>
  );
};
 
export default MilkControl;
