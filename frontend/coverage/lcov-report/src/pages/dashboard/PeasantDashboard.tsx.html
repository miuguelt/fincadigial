                          ? 'bg-emerald-500 text-slate-950 shadow-[0_10px_25px_-5px_rgba(16,185,129,0.4)]'
                          : 'bg-white/5 text-slate-400 border border-white/10 hover:bg-white/10'
                      }`}
                    >
                      {React.cloneElement(action.icon as React.ReactElement, { size: 14 })}
                      {action.title.split(' ')[1] || action.title}
                    </button>
                  ))}
                </div>
              </div>
 
              <CardContent className="p-12 space-y-10">
                <div className="grid gap-10">
                  <div className="space-y-4">
                    <label className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400 ml-6">{modeLabels[mode].primary}</label>
                    <Input
                      className="h-20 rounded-[1.75rem] border-slate-100 bg-slate-50/50 px-8 text-2xl font-black focus:bg-white focus:ring-4 focus:ring-emerald-500/10 transition-all border-2"
                      placeholder="Escribe aquí..."
                      value={primaryText}
                      onChange={(e) => setPrimaryText(e.target.value)}
                    />
                  </div>
                  
                  <div className="grid gap-10 md:grid-cols-2">
                    <div className="space-y-4">
                      <label className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400 ml-6">{modeLabels[mode].secondary}</label>
                      <Input
                        className="h-16 rounded-[1.5rem] border-slate-100 bg-slate-50/50 px-8 font-black text-lg focus:bg-white border-2"
                        placeholder="Info adicional"
                        value={secondaryText}
                        onChange={(e) => setSecondaryText(e.target.value)}
                      />
                    </div>
                    <div className="space-y-4">
                      <label className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400 ml-6">Observaciones</label>
                      <Textarea
                        className="min-h-16 max-h-16 rounded-[1.5rem] border-slate-100 bg-slate-50/50 px-8 py-4 font-bold focus:bg-white border-2"
                        placeholder="Detalles..."
                        value={notes}
                        onChange={(e) => setNotes(e.target.value)}
                      />
                    </div>
                  </div>
 
                  {feedback && (
                    <motion.div 
                      initial={{ opacity: 0, scale: 0.95 }}
                      animate={{ opacity: 1, scale: 1 }}
                      className="flex items-center gap-5 rounded-[2rem] bg-emerald-500/10 p-6 text-xs font-black text-emerald-600 border border-emerald-500/20 uppercase tracking-widest"
                    >
                      <div className="h-2.5 w-2.5 rounded-full bg-emerald-500 shadow-[0_0_10px_rgba(16,185,129,0.5)] animate-pulse" />
                      {feedback}
                    </motion.div>
                  )}
 
                  <Button
                    loading={saving}
                    onClick={submitQuickCapture}
                    className="h-24 w-full rounded-[2rem] bg-slate-900 text-xl font-black text-white shadow-2xl hover:bg-emerald-600 transition-all duration-500 uppercase tracking-[0.2em] group"
                  >
                    <Send className="mr-4 h-7 w-7 transition-transform group-hover:translate-x-2 group-hover:-translate-y-2" /> 
                    Guardar en Memoria
                  </Button>
                </div>
              </CardContent>
            </Card>
 
            <div className="rounded-[3rem] bg-white p-2 shadow-sm border border-slate-100">
               <NodeCommunicationWidget />
            </div>
          </div>
 
          {/* Sidebar de Monitoreo - Glass Overhaul */}
          <div className="space-y-8">
            <Card className="rounded-[3rem] border border-white/20 bg-white/40 dark:bg-slate-900/40 p-2 shadow-xl backdrop-blur-2xl">
               <CardContent className="p-8">
                  <div className="flex items-center gap-3 mb-6">
                    <RefreshCw className="h-4 w-4 text-emerald-500" />
                    <h4 className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400">Cola de Sincronización</h4>
                  </div>
                  <PendingSyncList />
               </CardContent>
            </Card>
 
            <Card className="rounded-[3rem] border border-white/20 bg-white/40 dark:bg-slate-900/40 p-2 shadow-xl backdrop-blur-2xl">
               <CardContent className="p-8">
                  <div className="flex items-center gap-3 mb-6">
                    <AlertTriangle className="h-4 w-4 text-rose-500" />
                    <h4 className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400">Anomalías de Crecimiento</h4>
                  </div>
                  <GrowthAnomalyMonitor />
               </CardContent>
            </Card>
 
            <Card className="rounded-[3rem] border border-white/20 bg-white/40 dark:bg-slate-900/40 p-2 shadow-xl backdrop-blur-2xl">
               <CardContent className="p-8">
                  <div className="flex items-center gap-3 mb-6">
                    <Signal className="h-4 w-4 text-blue-500" />
                    <h4 className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400">Estado de Red Mesh</h4>
                  </div>
                  <MeshMonitor />
               </CardContent>
            </Card>
 
            <div className="rounded-[2.5rem] bg-gradient-to-br from-emerald-500 to-emerald-700 p-8 text-white shadow-xl">
               <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-white/20 backdrop-blur mb-6">
                  <Signal className="h-6 w-6" />
               </div>
               <h4 className="text-xl font-black tracking-tight">Sincronización Mesh</h4>
               <p className="mt-2 text-sm font-medium text-white/80 leading-relaxed">
                 Tus datos se mueven entre dispositivos cercanos de forma invisible. La resiliencia rural está activa.
               </p>
            </div>
          </div>
        </section>
      </main>
 
      {/* Menú Flotante "Crystal Dock" */}
      <div className="fixed bottom-10 left-1/2 z-50 flex -translate-x-1/2 items-center gap-2 rounded-[2.5rem] border border-white/30 bg-slate-900/80 p-2 text-white shadow-[0_30px_80px_-15px_rgba(0,0,0,0.5)] backdrop-blur-2xl">
        {[
          { icon: <Leaf size={22} />, label: 'CULTIVOS', onClick: () => setMode('crop') },
          { icon: <Droplets size={22} />, label: 'AGUA', onClick: () => setMode('water') },
          { icon: <PackageOpen size={22} />, label: 'MERCADO', onClick: () => setMode('market') },
          { icon: <Megaphone size={22} />, label: 'AYUDA', onClick: () => setMode('assistance') },
          { icon: <PawPrint size={22} />, label: 'GANADO', onClick: () => setMode('animal') },
        ].map((item, idx) => (
          <button
            key={idx}
            onClick={item.onClick}
            className="group relative flex h-14 w-14 items-center justify-center rounded-2xl transition-all duration-300 hover:bg-emerald-500 hover:text-slate-950 hover:scale-110 active:scale-95"
          >
            {item.icon}
            <span className="absolute -top-14 left-1/2 -translate-x-1/2 rounded-xl bg-slate-900 border border-white/10 px-4 py-2 text-[9px] font-black uppercase tracking-[0.2em] text-white opacity-0 transition-all group-hover:-top-16 group-hover:opacity-100 shadow-2xl">
              {item.label}
            </span>
          </button>
        ))}
        <div className="mx-3 h-10 w-[1px] bg-white/20" />
        <motion.button 
          whileHover={{ rotate: 180, scale: 1.1 }}
          onClick={refreshData}
          className="flex h-14 w-14 items-center justify-center rounded-2xl bg-emerald-500 text-slate-950 shadow-[0_0_20px_rgba(16,185,129,0.3)] active:scale-95 transition-all"
        >
          <RefreshCw className={`h-6 w-6 ${saving ? 'animate-spin' : ''}`} />
        </motion.button>
      </div>
 
      {/* Estilos adicionales para consistencia */}
      <style dangerouslySetInnerHTML={{ __html: `
        .no-scrollbar::-webkit-scrollbar { display: none; }
        .no-scrollbar { -ms-overflow-style: none; scrollbar-width: none; }
      `}} />
    </div>
  );
};
